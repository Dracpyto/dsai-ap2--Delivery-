##### B. Módulo de Lógica (`src/js/search.js`)
Função pura de busca:
```javascript
export function normalizarTexto(texto) {
  if (!texto || typeof texto !== 'string') return '';
  return texto
    .toLowerCase()
    .trim()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '');
}

export function buscar(termo, restaurantes = [], menusData = {}) {
  const termoFormatado = normalizarTexto(termo);
  if (!termoFormatado) {
    return { termo: '', restaurantes: [], pratos: [], totalResultados: 0, encontrado: false };
  }

  const restaurantesEncontrados = restaurantes.filter(rest => {
    return normalizarTexto(rest.name).includes(termoFormatado) ||
           normalizarTexto(rest.category).includes(termoFormatado);
  });

  const pratosEncontrados = [];
  for (const rest of restaurantes) {
    const cardapio = menusData[rest.id] || [];
    for (const prato of cardapio) {
      if (normalizarTexto(prato.name).includes(termoFormatado) ||
          normalizarTexto(prato.description).includes(termoFormatado)) {
        pratosEncontrados.push({
          ...prato,
          restauranteId: rest.id,
          restauranteNome: rest.name
        });
      }
    }
  }

  const totalResultados = restaurantesEncontrados.length + pratosEncontrados.length;
  return {
    termo: termo.trim(),
    restaurantes: restaurantesEncontrados,
    pratos: pratosEncontrados,
    totalResultados,
    encontrado: totalResultados > 0
  };
}
```

### Gemini

#### Diagnóstico Técnico
O erro `EADDRINUSE` (Error: Address Already In Use) indica que a porta de rede solicitada (3000) já está sendo utilizada por outro processo no sistema operacional. No Node.js, quando o método `server.listen()` falha sem um ouvinte para o evento `'error'`, o runtime emite uma exceção não tratada (`Unhandled 'error' event`), provocando o encerramento imediato do processo.

Causas identificadas:
1. Uma instância anterior do próprio `server.js` em execução em segundo plano ou em outro terminal.
2. Outro serviço local (ex: React, Vite, Next.js) ocupando a porta 3000.

#### Solução Implementada em `server.js`
Foi refatorada a inicialização do servidor HTTP para:
1. Tratar o evento `'error'` com verificação do código `EADDRINUSE`.
2. Realizar **fallback automático para a próxima porta disponível** (`porta + 1`), garantindo que o servidor inicie sem quebrar a execução.
3. Permitir configuração flexível de porta via variável de ambiente `PORT` (`process.env.PORT || 3000`).

Trecho implementado em `server.js`:
```javascript
function iniciarServidor(porta) {
  const server = http.createServer(handleRequest);

  server.on('error', (err) => {
    if (err.code === 'EADDRINUSE') {
      console.warn(`⚠️  A porta ${porta} já está em uso por outro processo.`);
      const proximaPorta = porta + 1;
      console.log(`🔄 Tentando iniciar automaticamente na porta ${proximaPorta}...`);
      iniciarServidor(proximaPorta);
    } else {
      console.error('❌ Erro inesperado no servidor:', err);
    }
  });

  server.listen(porta, () => {
    console.log(`=========================================`);
    console.log(`🚀 Servidor DeliveryFast iniciado com sucesso!`);
    console.log(`👉 Abra no navegador: http://localhost:${porta}`);
    console.log(`=========================================`);
  });
}

iniciarServidor(INITIAL_PORT);
```
